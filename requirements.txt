# Python deps for rm-ocr. Everything else it uses is in the standard library
# (urllib, json, hashlib, pathlib, logging, ...).
#
# pdf2image rasterizes PDF pages; it shells out to poppler (pdftoppm / pdfunite),
# which is a SYSTEM package, NOT pip-installable:
#   Debian/Ubuntu : apt-get install poppler-utils
#   macOS         : brew install poppler
# pdf2image pulls in Pillow automatically.
pdf2image>=1.17

# pypdf reads the splitter's /RemarkableSplitter metadata marker and page sizes
# for the split-readiness gate (REQUIRE_SPLIT=1). Lazily imported, so only
# strictly required when that gate is enabled.
pypdf>=6.13.0

# PyMuPDF + numpy are used ONLY by AUTO_SPLIT (rm_split.py). PyMuPDF renders
# tall pages in-process for the whitespace-band analysis and assembles the
# split output by referencing each source page once (form XObject + clip), so
# a many-cut split of a vector-heavy export stays fast and roughly
# input-sized. The previous pypdf merge_page assembly re-encoded the full
# page content once per segment and could OOM on native reMarkable exports.
# Note PyMuPDF is AGPL-3.0. Both lazily imported, not needed unless AUTO_SPLIT=1.
PyMuPDF>=1.26
numpy>=2.4.6

# Pillow is pulled in by pdf2image anyway; pinned for the OCR image pipeline.
Pillow>=12.2.0

# inotify_simple powers the optional inotify wake-up signal (INOTIFY=1, on by
# default). Linux-only — on macOS the import will fail and the daemon falls back
# to pure polling. Pure-stdlib wrapper, no transitive deps.
inotify_simple>=2.0.1 ; sys_platform == "linux"


# rmc renders reMarkable .zip / .rmdoc / .rm bundles to PDF.
# The Scrybbling-together fork contains improvements to the original,
# including text-rendering fixes.
#
# It pulls in rmscene (the .rm parser) transitively.
# For PDF export, it uses CairoSVG to convert intermediate SVG
# pages to PDF, or Chrome/Chromium when available.
#
# Debian/Ubuntu: apt-get install libcairo2
#
# pdfunite (poppler-utils, installed above) merges the per-page PDFs.
rmc @ git+https://github.com/Scrybbling-together/rmc.git@main

# rmscene is a dependency of rmc and is therefore installed transitively.
# It is also a direct dependency of this project because rm_strokes.py
# imports it directly (STROKE_CONTEXT=1 - stroke-region hints, see README).
# We therefore declare it explicitly, using the same Scrybbling-together fork.
rmscene @ git+https://github.com/Scrybbling-together/rmscene.git@main
